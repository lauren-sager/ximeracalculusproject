\begin{exercise}   
  Consider the piecewise function 
  \[f(x)=
  \begin{cases}
    3x-7 & x\leq 0\\
    x^2+1 & 0<x<2\\
    4x-3 & x\geq 2
  \end{cases}\]
  \begin{enumerate}
    \item Sketch a graph of the function.
    \item Evaluate $\lim_{x\rightarrow 0^-}f(x)$.
    \textcolor{blue}{$=-7$}
    \item Evaluate $\lim_{x\rightarrow 0^+}f(x)$.
    \textcolor{blue}{$=1$}
    \item Evaluate $\lim_{x\rightarrow 0}f(x)$.
    \textcolor{blue}{does not exist (jump)}
    \item Evaluate $\lim_{x\rightarrow 2^-}f(x)$.
    \textcolor{blue}{$=5$}
    \item Evaluate $\lim_{x\rightarrow 2^+}f(x)$.
    \textcolor{blue}{$=5$}
    \item Evaluate $\lim_{x\rightarrow 2}f(x)$.
    \textcolor{blue}{$=5$}
  \end{enumerate}
\end{exercise}

\begin{exercise}
    \textbf{Challenge Problem} Write an \textit{equation} of a function, $f$, that satisfies the following conditions.  (Hint: A piecewise function may be helpful here.)
    \begin{enumerate}
        \item $\displaystyle\lim_{x\rightarrow -2^-}f(x)=-2$
        \item $\displaystyle \lim_{x\rightarrow -2^+}f(x)=3$
        \item $\displaystyle \lim_{x\rightarrow 1}f(x)=0$
        \item $f(-2)=3$
        \item $f(1)\neq 0$
    \end{enumerate}

    \textcolor{blue}{For example, \[f(x)=\begin{cases}
    x & x<-2\\
    -x+1 & -2\leq x<1\\
    -4 & x=1\\
    x-1 & x>1
    \end{cases}\]}
    \textcolor{blue}{is a simple answer (the function need not change at $x=1$, as long as a separate point is defined at $x=1$).  Many other examples are possible!}
\end{exercise}
