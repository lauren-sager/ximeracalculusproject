\begin{exercise}
    Sketch the graph of a function $g(x)$ that satisfies all of the following conditions: \\
    \begin{tabular}{c|c|c|c}
       Interval  & $g$ & $g'$ & $g''$  \\ \hline
        $(-\infty, -1)$ &  & $g'(x)>0$ & $g''(x)>0$\\
        x=-1 & 0 & DNE & DNE \\
        $(-1, 3)$ & & $g'(x)<0$ & $g''(x)>0$\\
        x=3 & -3 & 0 & $g''(x)>0$\\
        $(3,5)$ & & $g'(x)>0$ & $g''(x)>0$\\
        x=5 & 4 & $g'(x)>0 $& 0\\
        $(5,\infty)$ & & $g'(x)>0$ & $g''(x)<0$
    \end{tabular}
\end{exercise}

\end{exercise}
