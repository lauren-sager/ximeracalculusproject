\begin{exercise}
    On a coordinate plane, draw a function, $f$, that satisfies the following conditions.  Be neat in your drawing, and label any important points.
        \begin{enumerate}
            \item $\displaystyle\lim_{x\rightarrow -1^+}f(x)=-2$
            \item $\displaystyle\lim_{x\rightarrow -1^-} f(x)=3$
            \item $f(x)$ has domain $[-5, -1)\cup(-1,7] $
            \item $\displaystyle\lim_{x\rightarrow 2}f(x)=0$
            %\item $f(x)$ has a removable discontinuity at $x=2$.
            %\item $f(x)$ is left continuous at $x=7$.
            %\item $\displaystyle\lim_{x\rightarrow 4}f(x)=-\infty$.
            \item $\displaystyle\lim_{x\rightarrow -3}f(x)=4$
            \item $\displaystyle\lim_{x\rightarrow 7^-} f(x)=-3=f(7)$
            \item $f(2)\neq 0$
        \end{enumerate}
\end{exercise}

\begin{exercise}
    Use the given table to evaluate the following limits, or explain why you cannot. \\
  \begin{center}
        \begin{tabular}{c|c|c}
            $c$ & $\lim_{x\rightarrow c}f(x)$ & $\lim_{x\rightarrow c}g(x)$  \\ \hline
            -2 &  5 & 3\\ 
            0 & -1 & DNE\\ 
            2 & $\displaystyle\frac{1}{2}$ & -2\\
            3 & 0 & 4
        \end{tabular}
     \end{center}
     \begin{enumerate}
     \item $\displaystyle\lim_{x\rightarrow -2} f(x)(g(x))^2$
     \item $\displaystyle\lim_{x\rightarrow 0} \frac{g(x)}{f(x)}$
     \item $\displaystyle \lim_{x\rightarrow 0} \sqrt[3]{f(x)}$
     \item $\displaystyle \lim_{x\rightarrow 2} 3f(x)-2g(x)+7x$ 
     \item $\displaystyle \lim_{x\rightarrow 3} \frac{2f(x)+1}{\sqrt{g(x)}}$ 
    \end{enumerate}
\end{exercise}
